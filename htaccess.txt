## Apache rules for sudoshz.ir
## Rename to .htaccess on the host if Apache is in use.

RewriteEngine On

## HTTPS
RewriteCond %{HTTPS} !=on
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

## www → apex
RewriteCond %{HTTP_HOST} ^www\.(.+)$ [NC]
RewriteRule ^ https://%1%{REQUEST_URI} [L,R=301]

## Never expose Publii publish folder
RewriteRule ^main-sudoshz/(.*)$ /$1 [L,R=301]

## index.html → /
RewriteRule ^index\.html$ / [L,R=301]

## Trailing slash for extensionless URLs (direct, no internal folder leak)
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_URI} !/$
RewriteCond %{REQUEST_URI} !\.[a-zA-Z0-9]{1,8}$
RewriteRule ^(.*)$ /$1/ [L,R=301]
